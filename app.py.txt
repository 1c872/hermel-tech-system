def init_db():
    conn = sqlite3.connect(DB_FILE)
    cursor = conn.cursor()
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS teachers (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            empId TEXT UNIQUE,
            name TEXT NOT NULL,
            specialty TEXT,
            position TEXT,
            assignedHours INTEGER,
            phone TEXT,
            status TEXT DEFAULT 'فعّال',
            username TEXT UNIQUE,
            password TEXT,
            role TEXT DEFAULT 'teacher',
            subjects TEXT
        )
    ''')
    cursor.execute("SELECT * FROM teachers WHERE username = 'admin'")
    if not cursor.fetchone():
        cursor.execute("INSERT INTO teachers (empId, name, specialty, position, assignedHours, phone, username, password, role, subjects) VALUES ('1', 'مدير المعهد', 'إدارة عامة', 'مدير المعهد', 0, '03000000', 'admin', '123456', 'admin', 'الإدارة')")
        sample_teachers = [

    conn.commit()
    conn.close()

@app.route('/api/teachers', methods=['GET'])
def get_teachers():
    conn = sqlite3.connect(DB_FILE)
    conn.row_factory = sqlite3.Row
    cursor = conn.cursor()
    teachers = [dict(row) for row in cursor.execute("SELECT * FROM teachers").fetchall()]
    conn.close()
    return jsonify(teachers)

@app.route('/api/teachers', methods=['POST'])

    data = request.json

    cursor = conn.cursor()
    try:
        cursor.execute("""
            INSERT INTO teachers (empId, name, specialty, position, assignedHours, phone, username, password, role, subjects)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, 'teacher', ?)
        """, (data.get('empId'), data.get('name'), data.get('specialty'), data.get('position', 'أستاذ مادة'), data.get('assignedHours', 0), data.get('phone', ''), data.get('username'), data.get('password', '123456'), data.get('subjects', '')))
        conn.commit()
        success = True
    except:
        success = False
    conn.close()
    return jsonify({"success": success})

@app.route('/api/teachers/<int:teacher_id>', methods=['DELETE'])
def delete_teacher(teacher_id):
    conn = sqlite3.connect(DB_FILE)
    cursor = conn.cursor()
    cursor.execute("DELETE FROM teachers WHERE id = ? AND role != 'admin'", (teacher_id,))
    conn.commit()
    conn.close()
    return jsonify({"success": True})
